script> let audioContext=null, analyser=null, mic=null; let dataArray, animationId=null; let scrollPos=0, sensitivity=4; let silenceTimer=0, silenceLimit=30; let isPaused=false, mirrored=false; let playIndex=0; if(speechEnergy<200){ silenceTimer++; if(silenceTimer>silenceLimit) isPaused=true; }else{ silenceTimer=0; isPaused=false; } if(!isPaused){ let speed=(speechEnergy/800)*sensitivity; scrollPos+=speed; $("#content").css("top",-scrollPos+"px"); } animationId=requestAnimationFrame(animateScroll); } /* CONTROLS */ $("#startMic").click(()=>{ $("#content").text($("#scriptText").val()); scrollPos=0; startMic(); }); $("#fontSize").on("input",e=>{ $("#content").css("font-size",e.target.value+"px"); $("#content").css("transform",mirrored?"scaleX(-1)":"scaleX(1)"); }); /* STORAGE */ function loadScripts(){ let s=JSON.parse(localStorage.getItem("teleScripts")||"{}"); $("#playlist").empty(); $.each(s,k=>$("#playlist").append(``));
